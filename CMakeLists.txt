cmake_minimum_required(VERSION 3.16)
project(parallel_ismcts CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(TSAN "Build with ThreadSanitizer" OFF)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -march=native")
if(TSAN)
  set(CMAKE_CXX_FLAGS_RELEASE "-O1 -g -fsanitize=thread -DPM_STD_THREAD -DNDEBUG")
else()
  set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
  find_package(OpenMP REQUIRED)
  link_libraries(OpenMP::OpenMP_CXX)
endif()
find_package(Threads REQUIRED)
link_libraries(Threads::Threads)
include_directories(src)

enable_testing()
file(GLOB TEST_SRCS CONFIGURE_DEPENDS tests/test_*.cpp)
foreach(src ${TEST_SRCS})
  get_filename_component(name ${src} NAME_WE)
  add_executable(${name} ${src})
  add_test(NAME ${name} COMMAND ${name})
endforeach()
if(TSAN)
  add_executable(tsan_negative_control tests/tsan_negative_control.cpp)
endif()
file(GLOB MAIN_SRCS CONFIGURE_DEPENDS src/main_*.cpp)
foreach(src ${MAIN_SRCS})
  get_filename_component(name ${src} NAME_WE)
  add_executable(${name} ${src})
endforeach()
